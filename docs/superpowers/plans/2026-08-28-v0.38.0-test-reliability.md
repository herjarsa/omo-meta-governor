**Goal:** Reduce skipped test count from 12 to 2 by adding a global `uncaughtException` handler and root-cause fixing the 3 Windows-flaky tests re-quarantined in v0.37.2.

**Architecture:** A new `src/error-handler.ts` installs a `process.on('uncaughtException')` filter that swallows chokidar/readdirp errors from scanning system-protected paths (`D:\pagefile.sys`, etc.) and re-throws unknown errors. **Tech Stack:** TypeScript, Bun, existing chokidar dep, existing `MetaGovernorPluginDeps` DI seam.

**Files:**

**Interfaces:**

- Produces: working filter that swallows chokidar/readdirp errors and re-throws unknowns

- [ ] **Step 1: Replace the function body with the real implementation**

```typescript
import type { Logger } from "./logger"

export const DEFAULT_PATH_PATTERNS: RegExp[] = [
  /^[A-Z]:\\(pagefile\.sys|DumpStack\.log\.tmp)/i,
  /^\/(var|private\/var)\/folders\//,
  /^\/snap\//,
]

export const DEFAULT_ERROR_CODES = new Set(["EINVAL", "ENOENT", "EACCES"])

export interface ErrorHandlerOptions {
  logger?: Logger
  paths?: RegExp[]
  errorCodes?: Set<string>
}

export function installGlobalErrorHandler(opts: ErrorHandlerOptions = {}): () => void {
  const paths = opts.paths ?? DEFAULT_PATH_PATTERNS
  const errorCodes = opts.errorCodes ?? DEFAULT_ERROR_CODES
  const logger = opts.logger ?? console
  const previousHandler = process.listeners("uncaughtException").at(-1)
  const handler = (err: Error & { code?: string; path?: string }) => {
    const isWatcherError =
      err && typeof err === "object" && "code" in err && err.code
        ? errorCodes.has(err.code)
        : false
    const isSystemPath = !!(err?.path && paths.some((p) => p.test(err.path!)))
    if (isWatcherError && isSystemPath) {
      logger.warn?.("watcher_scan_blocked", {
        path: err.path,
        code: err.code,
        message: err.message,
      })
      return
    }
    if (previousHandler) {
      ;(previousHandler as (e: Error) => void)(err)
    } else {
      throw err
    }
  }
  process.on("uncaughtException", handler)
  return () => process.off("uncaughtException", handler)
}
```
