# v0.38.0 Schema Auto-Generation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace hand-written `src/generate-schema.ts` with auto-generation from TypeScript types using `ts-json-schema-generator@2.9.0`. Schema drift becomes structurally impossible.

**Architecture:** `src/config.ts` becomes the single source of truth. A new `src/generate-schema.ts` (rewritten) uses `ts-json-schema-generator` to parse the `MetaGovernorPluginConfig` interface, post-processes the output to inject runtime-known defaults (URLs, paths), and writes the JSON to `assets/omo-meta-governor.schema.json`. The build pipeline stays unchanged.

**Tech Stack:** TypeScript, Bun, `ts-json-schema-generator@2.9.0` (new devDep), JSDoc annotations for defaults.

**Spec:** `docs/superpowers/specs/2026-08-28-v0.38.0-schema-autogen-design.md`

## Global Constraints

- TypeScript strict mode (no `any`, no `@ts-ignore`)
- All commits must be atomic per AGENTS.md §1
- CI must be green on all 3 platforms (macOS, Windows, Ubuntu) before merge
- Public API surface preserved: `generateSchema()` and `writeSchemaFile()` signatures unchanged
- `src/generate-schema-sync.test.ts` (added in v0.37.2) must keep passing
- Add `@default` JSDoc tags to `src/config.ts` for ALL fields with runtime defaults (~40 tags total)

## Task 5: Rewrite `src/generate-schema.ts` with `ts-json-schema-generator`

**Files:**
- Modify: `src/generate-schema.ts` (rewrite)

**Interfaces:**
- Consumes: `MetaGovernorPluginConfig` from `src/config.ts`
- Produces: `generateSchema(): JsonSchema`, `writeSchemaFile(path: string): Promise<void>`, `JsonSchema` type, `JsonSchemaProperty` type

- [ ] **Step 1: Replace `src/generate-schema.ts` with the new implementation**

```typescript
import { createGenerator, type Config, type SchemaGenerator } from "ts-json-schema-generator"
import { resolve } from "node:path"

export interface JsonSchema {
  $schema: string
  $id: string
  title: string
  description: string
  type: "object"
  properties: Record<string, JsonSchemaProperty>
  additionalProperties: boolean
  definitions?: Record<string, JsonSchemaProperty>
  [key: string]: unknown
}

export interface JsonSchemaProperty {
  type?: string | string[]
  description?: string
  default?: unknown
  properties?: Record<string, JsonSchemaProperty>
  items?: JsonSchemaProperty
  additionalProperties?: boolean
  required?: string[]
  enum?: string[]
  oneOf?: JsonSchemaProperty[]
  anyOf?: JsonSchemaProperty[]
  $ref?: string
  minimum?: number
  maximum?: number
  [key: string]: unknown
}

const TYPESCRIPT_CONFIG: Config = {
  path: resolve(import.meta.dir, "config.ts"),
  tsconfig: resolve(import.meta.dir, "..", "tsconfig.json"),
  type: "MetaGovernorPluginConfig",
  schemaId: "auto",
  topRef: false,
  skipTypeCheck: true,
  encodeRefs: false,
  additionalProperties: false,
  discriminatorMapping: {},
}

let _generator: SchemaGenerator | null = null
function getGenerator(): SchemaGenerator {
  if (!_generator) _generator = createGenerator(TYPESCRIPT_CONFIG)
  return _generator
}

function generateRawSchema(): JsonSchema {
  const gen = getGenerator()
  return gen.createSchema("MetaGovernorPluginConfig") as JsonSchema
}

function applyRuntimeDefaults(raw: JsonSchema): JsonSchema {
  const out = structuredClone(raw)
  const skillHub = out.properties?.skillHub?.properties
  if (skillHub) {
    if (skillHub.bootstrapUrl && !skillHub.bootstrapUrl.default) {
      skillHub.bootstrapUrl.default = "https://skills-library.com/api/skills.json"
    }
  }
  const cliAnything = out.properties?.cliAnything?.properties
  if (cliAnything) {
    if (cliAnything.cliHubBin && !cliAnything.cliHubBin.default) {
      cliAnything.cliHubBin.default = "cli-hub"
    }
    if (cliAnything.skillsBin && !cliAnything.skillsBin.default) {
      cliAnything.skillsBin.default = "npx skills"
    }
  }
  return out
}

function enforceAdditionalPropertiesFalse(node: JsonSchemaProperty | JsonSchema | undefined): void {
  if (!node || typeof node !== "object") return
  if (node.type === "object" && node.properties && !("$ref" in node)) {
    if (node.additionalProperties === undefined) {
      node.additionalProperties = false
    }
    for (const key of Object.keys(node.properties)) {
      enforceAdditionalPropertiesFalse(node.properties[key])
    }
  }
  if (node.items) enforceAdditionalPropertiesFalse(node.items)
}

function finalizeSchema(raw: JsonSchema): JsonSchema {
  const out = structuredClone(raw)
  enforceAdditionalPropertiesFalse(out)
  out.properties ??= {}
  out.properties.$schema = {
    type: "string",
    description: "JSON Schema reference for IDE autocompletion.",
  }
  const sorted: Record<string, JsonSchemaProperty> = {}
  for (const key of Object.keys(out.properties).sort()) {
    sorted[key] = out.properties[key]
  }
  out.properties = sorted
  return out
}

export function generateSchema(): JsonSchema {
  const raw = generateRawSchema()
  const withDefaults = applyRuntimeDefaults(raw)
  return finalizeSchema(withDefaults)
}

export async function writeSchemaFile(outputPath: string): Promise<void> {
  const schema = generateSchema()
  const { writeFile, rename, mkdtemp, rm } = await import("node:fs/promises")
  const { tmpdir } = await import("node:os")
  const tmpDir = await mkdtemp(resolve(tmpdir(), "schema-"))
  const tmpFile = resolve(tmpDir, "schema.json")
  try {
    await writeFile(tmpFile, JSON.stringify(schema, null, 2) + "\n", "utf-8")
    await rename(tmpFile, outputPath)
  } finally {
    await rm(tmpDir, { recursive: true, force: true }).catch(() => {})
  }
}
```
